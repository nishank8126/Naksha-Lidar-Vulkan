def _normalized_delivery_project_stem(filename: str) -> str:
    """Return a conservative project identity for classified SNT copies."""
    stem = Path(str(filename)).stem.strip()
    return re.sub(
        r"(?:[ _-]+class(?:ification)?(?:[ _-]*\d+)?)$",
        "",
        stem,
        flags=re.IGNORECASE,
    ).strip(" _-").casefold()

    # Classified/edited SNT exports may contain no grid polygons, or only a few
    # incidental polygons that spatially match unrelated PRJ tiles. The PRJ is
    # the authoritative complete grid/file index, so merge every distinct PRJ
    # block instead of using it only when the SNT selection is completely empty.
    if prj_blocks:
        prj_names = {
            str(block.get("label") or "").strip().casefold()
            for block in prj_blocks
            if str(block.get("label") or "").strip()
        }
        # Replace any approximate/incidental SNT entry carrying a PRJ label;
        # the PRJ boundary and its direct source path are authoritative.
        app.snt_block_polygons = [
            entry for entry in app.snt_block_polygons
            if not (
                entry.get("snt_filename") == snt_filename
                and str(entry.get("grid_name") or "").strip().casefold() in prj_names
            )
        ]
        indexed_prj_names = set()
        retained_prj_count = 0
        for block in prj_blocks:
            points_xy = _open_polygon_xy(block.get("points_2d") or [])
            grid_name = str(block.get("label") or "").strip()
            if len(points_xy) < 3 or not grid_name:
                continue
            normalized_grid_name = grid_name.casefold()
            if normalized_grid_name in indexed_prj_names:
                continue

            block_file = str(block.get("block_file") or "").strip()
            alt_names = []
            if block_file:
                block_file_stem = Path(block_file).stem
                if block_file_stem and block_file_stem.casefold() != grid_name.casefold():

            app.snt_block_polygons.append({
                "points_2d": [(float(x), float(y)) for x, y in points_xy],
                "grid_name": grid_name,
                "block_file": block_file,
                "alt_names": alt_names,
                "file_path": (

                ),
                "source": "prj_authority",
                "snt_filename": snt_filename,
                "prj_path": block.get("prj_path"),
                "poly_layer": "PRJ",

            indexed_prj_names.add(normalized_grid_name)
            retained_prj_count += 1

        print(
            f"[SNT block polygons] retained {retained_prj_count} distinct "
            f"PRJ authority block(s) alongside {len(bl_polys)} SNT polygon(s)"
