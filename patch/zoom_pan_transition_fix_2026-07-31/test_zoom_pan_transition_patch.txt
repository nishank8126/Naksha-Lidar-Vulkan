from types import SimpleNamespace

from gui.app_window import NakshaApp
from gui.gpu_render_manager import GPURenderManager


class _FakeTimer:
    def __init__(self):
        self.stop_count = 0

    def stop(self):
        self.stop_count += 1


def test_middle_pan_cancels_pending_zoom_without_changing_camera():
    history = []
    timer = _FakeTimer()
    app = SimpleNamespace(
        _zoom_anim_active=True,
        _zoom_target_factor=1.1,
        _zoom_anim_timer=timer,
        _schedule_main_view_history_commit=lambda reason, delay_ms: history.append(
            (reason, delay_ms)
        ),
    )

    assert NakshaApp._cancel_smooth_zoom_for_pan(app)
    assert app._zoom_target_factor == 1.0
    assert not app._zoom_anim_active
    assert timer.stop_count == 1
    assert history == [("smooth_mouse_wheel_zoom", 80)]


def test_render_manager_transitions_wheel_to_pan_and_settles_release():
    calls = []
    wheel_timer = _FakeTimer()
    app = SimpleNamespace(
        _shutdown_in_progress=False,
        _cancel_smooth_zoom_for_pan=lambda: calls.append("cancel_zoom"),
    )
    manager = SimpleNamespace(
        _wheel_idle_timer=wheel_timer,
        _wheel_interaction_active=True,
        _interaction_active=True,
        _pan_in_progress=False,
        app=app,
        _engage_lod=lambda: calls.append("engage"),
        _restore_full_detail=lambda: calls.append("restore"),
        _execute_render=lambda: calls.append("render"),
    )

    GPURenderManager._on_pan_start(manager, None, None)
    assert manager._pan_in_progress
    assert manager._interaction_active
    assert not manager._wheel_interaction_active
    assert calls == ["cancel_zoom", "engage"]

    GPURenderManager._on_pan_end(manager, None, None)
    assert not manager._pan_in_progress
    assert not manager._interaction_active
    assert not manager._wheel_interaction_active
    assert wheel_timer.stop_count == 2
    assert calls[-2:] == ["restore", "render"]
