    def _find_matching_las_file(self, las_folder, grid_name):
        """
        Find LAZ/LAS file matching grid name using a strict scored match.

        Preference order:
        - exact full stem match
        - exact numeric suffix match
        - suffix-ending stem match
        - normalized stem match
        """
        print(f"\n🔍 Searching for file matching: {grid_name}")
        
        las_folder_path = Path(las_folder)
        
        # Get all LAZ/LAS files
        all_files = list(las_folder_path.glob("*.laz")) + list(las_folder_path.glob("*.las"))
        
        print(f"   Total files in folder: {len(all_files)}")
        
        if not all_files:
            return None

        def _norm(text: str) -> str:
            return "".join(ch.lower() for ch in str(text) if ch.isalnum())

        def _suffix_digits(text: str) -> str:
            m = re.search(r"(\d+)(?!.*\d)", str(text))
            return m.group(1) if m else ""

        def _strip_las_ext(name: str) -> str:
            """Strip .laz/.las extension from name if present."""
            lower = name.lower()
            if lower.endswith(".laz"):
                return name[:-4]
            if lower.endswith(".las"):
                return name[:-4]
            return name

        # Strip extension from grid_name to avoid mismatch with file stems
        grid_name_clean = _strip_las_ext(grid_name)
        grid_norm = _norm(grid_name_clean)
        grid_suffix = _suffix_digits(grid_name_clean)
        scores = []

        for file_path in all_files:
            stem = file_path.stem
            stem_norm = _norm(stem)
            stem_suffix = _suffix_digits(stem)

            score = 0
            if stem.lower() == grid_name_clean.lower():
                score += 2000
            if stem_norm == grid_norm:
                score += 1500
            if grid_suffix and stem_suffix == grid_suffix:
                score += 1200
            if grid_suffix and stem_norm.endswith(grid_suffix):
                score += 800
            if grid_norm and stem_norm.endswith(grid_norm[-8:]):
                score += 200
            if grid_norm and grid_norm in stem_norm:
                score += 100
            if grid_norm and stem_norm in grid_norm:
                score += 50

            if score > 0:
                scores.append((score, len(stem_norm), file_path))

        if not scores:
            print(f"   ❌ No match found for '{grid_name_clean}' (norm='{grid_norm}', suffix='{grid_suffix}')")
            print(f"   📄 Available files:")
            for fp in all_files[:10]:
                print(f"      - {fp.name} (stem_norm='{_norm(fp.stem)}', suffix='{_suffix_digits(fp.stem)}')")
            return None

        scores.sort(key=lambda x: (-x[0], x[1], x[2].name.lower()))
        best = scores[0][2]
        print(f"   ✅ MATCH FOUND: {best.name} (score={scores[0][0]})")
        if len(scores) > 1:
            preview = ", ".join(f"{item[2].name}(s={item[0]})" for item in scores[:5])
            print(f"   ℹ️ Ranked candidates: {preview}")
        return best

    def _find_matching_las_file(self, las_folder, grid_name):
        """
        Find the unique LAZ/LAS file with the same complete grid identity.

        A partial coordinate (for example, matching only the northing) is never
        accepted. Missing or ambiguous matches return None so the caller opens
        the owning folder instead of loading a neighbouring grid.
        """
        print(f"\n🔍 Searching for file matching: {grid_name}")
        
        las_folder_path = Path(las_folder)
        
        # Get all LAZ/LAS files
        all_files = list(las_folder_path.glob("*.laz")) + list(las_folder_path.glob("*.las"))
        
        print(f"   Total files in folder: {len(all_files)}")
        
        if not all_files:
            return None

        grid_name_clean = strip_lidar_extension(grid_name)
        best = find_matching_lidar_file(all_files, grid_name_clean)

        if best is None:
            print(
                f"   ❌ No unique full-identity match for '{grid_name_clean}' "
                f"(norm='{normalized_stem(grid_name_clean)}', "
                f"numbers={numeric_identity(grid_name_clean)})"
            )
            print(f"   📄 Available files:")
            for fp in all_files[:10]:
                print(
                    f"      - {fp.name} "
                    f"(norm='{normalized_stem(fp.stem)}', "
                    f"numbers={numeric_identity(fp.stem)})"
                )
            return None

        print(f"   ✅ EXACT GRID MATCH FOUND: {best.name}")
        return best
