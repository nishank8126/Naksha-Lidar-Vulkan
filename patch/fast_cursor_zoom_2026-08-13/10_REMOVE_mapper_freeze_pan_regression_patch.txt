    def _freeze_main_pan_pipelines(self) -> int:
        """Temporarily mark visible main-view mappers static during camera-only pan."""
        if self._pan_static_mappers:
            return len(self._pan_static_mappers)

        app = self.app
        vtk_widget = getattr(app, "vtk_widget", None) if app is not None else None
        if vtk_widget is None:
            return 0

        frozen = []
        seen = set()
        try:
            render_window = vtk_widget.GetRenderWindow()
            renderers = render_window.GetRenderers() if render_window is not None else None
            if renderers is None:
                return 0

            renderers.InitTraversal()
            renderer_count = int(renderers.GetNumberOfItems())
            for _ in range(renderer_count):
                renderer = renderers.GetNextItem()
                if renderer is None:
                    continue
                actors = renderer.GetActors()
                if actors is None:
                    continue
                actors.InitTraversal()
                actor_count = int(actors.GetNumberOfItems())
                for _ in range(actor_count):
                    actor = actors.GetNextActor()
                    if actor is None:
                        continue
                    try:
                        if hasattr(actor, "GetVisibility") and not actor.GetVisibility():
                            continue
                        mapper = actor.GetMapper()
                        if mapper is None or not hasattr(mapper, "SetStatic"):
                            continue
                        mapper_id = id(mapper)
                        if mapper_id in seen:
                            continue
                        seen.add(mapper_id)
                        original_static = int(mapper.GetStatic()) if hasattr(mapper, "GetStatic") else 0
                        frozen.append((mapper, original_static))
                        if not original_static:
                            mapper.SetStatic(1)
                    except (RuntimeError, AttributeError, OSError, ReferenceError):
                        continue
        except (RuntimeError, AttributeError, OSError, ReferenceError):
            pass
        except Exception:
            pass

        self._pan_static_mappers = frozen
        return len(frozen)

    def _restore_main_pan_pipelines(self) -> int:
        """Restore every mapper flag captured by _freeze_main_pan_pipelines."""
        frozen = self._pan_static_mappers
        self._pan_static_mappers = []
        restored = 0
        for mapper, original_static in frozen:
            try:
                mapper.SetStatic(int(original_static))
                restored += 1
            except (RuntimeError, AttributeError, OSError, ReferenceError):
                pass
            except Exception:
                pass
        return restored
